name: Notificador de tickets

on:
  schedule:
    - cron: "*/10 * * * *"   # cada 10 minutos (GitHub puede retrasarlo unos minutos)
  workflow_dispatch:          # botón para correrlo a mano

concurrency:
  group: notificador
  cancel-in-progress: false

jobs:
  revisar:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
          cache: pip

      - name: Instalar dependencias
        run: |
          pip install -r requirements.txt
          python -m playwright install --with-deps chromium

      # Recupera la lista de tickets ya vistos de la ejecución anterior
      - name: Restaurar estado
        uses: actions/cache/restore@v4
        with:
          path: state
          key: tickets-state-${{ github.run_id }}
          restore-keys: tickets-state-

      - name: Revisar portal
        env:
          PORTAL_EMAIL: ${{ secrets.PORTAL_EMAIL }}
          PORTAL_PASSWORD: ${{ secrets.PORTAL_PASSWORD }}
          NTFY_TOPIC: ${{ secrets.NTFY_TOPIC }}
          FILTRO: ${{ secrets.FILTRO }}
          PORTAL_URL: ${{ secrets.PORTAL_URL }}
        run: python monitor.py

      - name: Guardar estado
        if: success()
        uses: actions/cache/save@v4
        with:
          path: state
          key: tickets-state-${{ github.run_id }}
